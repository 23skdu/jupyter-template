Jupyter Server is starting. Watch it come up with:
  kubectl --namespace {{ .Release.Namespace }} rollout status deployment/{{ $fullName }}

1. Reach the notebook server:
{{- if .Values.httpRoute.enabled }}
  {{- /* Resolve the first configured path, defaulting to "/". Guarded against
  nil lists so a partially configured httpRoute block still renders. */ -}}
  {{- $path := "/" -}}
  {{- $rules := .Values.httpRoute.rules | default list -}}
  {{- if $rules -}}
    {{- $matches := (first $rules).matches | default list -}}
    {{- if $matches -}}
      {{- with (first $matches).path -}}
        {{- with .value -}}{{- $path = . -}}{{- end -}}
      {{- end -}}
    {{- end -}}
  {{- end -}}
  {{- $parent := first (.Values.httpRoute.parentRefs | default list) }}
  {{- if .Values.httpRoute.hostnames }}
    {{- range .Values.httpRoute.hostnames }}
    # Scheme and TLS come from the parent Gateway's listener, not from this chart.
    export APP_URL="{{ . }}{{ $path }}"
    {{- end }}
  {{- else }}
    {{- with $parent }}
    export APP_HOSTNAME=$(kubectl get --namespace {{ .namespace | default $.Release.Namespace }} gateway/{{ .name }} -o jsonpath="{.spec.listeners[0].hostname}")
    echo "Visit http://$APP_HOSTNAME{{ $path }} to use your application"
    {{- end }}
  {{- end }}
  {{- with $parent }}
  NOTE: the route only serves what the parent Gateway's listeners and your
  HTTPRoute rules allow. Inspect the Gateway with:
    kubectl get --namespace {{ .namespace | default $.Release.Namespace }} gateway/{{ .name }} -o yaml
  {{- end }}
{{- else if .Values.ingress.enabled }}
{{- range $host := .Values.ingress.hosts }}
  {{- range .paths }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}{{ .path }}
  {{- end }}
{{- end }}
  NOTE: Jupyter keeps long-lived websockets open for terminals and kernels. If
  your controller times out idle connections, raise the proxy read/send timeout
  via ingress annotations (for nginx-ingress, proxy-read-timeout / -send-timeout).
{{- else if eq "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ $fullName }})
  export NODE_IP=$(kubectl get nodes -o jsonpath="{.items[0].status.addresses[0].address}")
  echo "Visit http://$NODE_IP:$NODE_PORT to use your application"
{{- else if eq "LoadBalancer" .Values.service.type }}
     NOTE: it may take a few minutes for the LoadBalancer IP to be assigned.
           Watch it with:
           kubectl get --namespace {{ .Release.Namespace }} svc -w {{ $fullName }}
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ $fullName }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  export SERVICE_PORT=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ $fullName }} -o jsonpath="{.spec.ports[0].port}")
  echo "Visit http://$SERVICE_IP:$SERVICE_PORT to use your application"
{{- else }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "jupyter-template.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  kubectl --namespace {{ .Release.Namespace }} port-forward service/{{ $fullName }} 8080:{{ .Values.service.port }}
  echo "Visit http://127.0.0.1:8080 to use your application"
{{- end }}

2. Log in with the token:
{{- if and .Values.token.injectEnv .Values.token.create (not .Values.token.existingSecret) }}
  kubectl --namespace {{ .Release.Namespace }} get secret {{ $fullName }} \
    -o jsonpath='{.data.token}' | base64 -d; echo
  Then open the URL printed above and paste the token into the login form.
  NOTE: leave token.value empty and Helm generates a random token on first
  install, then keeps it stable across upgrades. Setting token.value is only
  needed if you want a token you know in advance.
{{- else if and .Values.token.injectEnv .Values.token.existingSecret }}
  kubectl --namespace {{ .Release.Namespace }} get secret {{ .Values.token.existingSecret }} \
    -o jsonpath='{.data.{{ .Values.token.existingSecretKey }}}' | base64 -d; echo
{{- else if .Values.token.injectEnv }}
  WARNING: token.injectEnv is enabled but neither token.create nor
  token.existingSecret is set, so no JUPYTER_TOKEN was injected and Jupyter
  generated a random one at startup. Set token.existingSecret or turn off
  token.injectEnv.
{{- else }}
  No JUPYTER_TOKEN is injected (token.injectEnv=false). Configure auth with
  extraArgs, for example a hashed password:
    extraArgs:
      - --PasswordIdentityProvider.hashed_password=<hash>
{{- end }}
